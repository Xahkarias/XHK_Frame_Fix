
engineer_is_nanotech_researcher_trigger = {

        frameworld_has_active_nanite_upgrade = yes
        has_active_building = building_nanolab_1
        has_active_building = building_nanolab_2
    }

can_embark_trigger = {
	has_nomads_dlc = yes
	is_nomadic = no
	OR = {
		has_tradition = tr_adaptability_finish
		has_tradition = tr_versatility_finish
		has_country_flag = doomsday_embark_unlocked
	}
	NOT = {
		has_origin = origin_frameworld
	}
	can_embark_no_tradition_check_trigger = yes
}
